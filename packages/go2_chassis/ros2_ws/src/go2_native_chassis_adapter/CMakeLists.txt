cmake_minimum_required(VERSION 3.16)
project(go2w_native_chassis_adapter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(unitree_go REQUIRED)
add_executable(go2w_native_chassis_adapter_node src/native_adapter.cpp)
target_include_directories(go2w_native_chassis_adapter_node PRIVATE ../../../include)
target_compile_options(go2w_native_chassis_adapter_node PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(go2w_native_chassis_adapter_node
  rclcpp geometry_msgs diagnostic_msgs std_srvs unitree_go)
install(TARGETS go2w_native_chassis_adapter_node DESTINATION lib/${PROJECT_NAME})
add_executable(go2w_native_state_probe src/state_probe.cpp)
target_include_directories(go2w_native_state_probe PRIVATE ../../../include)
target_compile_options(go2w_native_state_probe PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(go2w_native_state_probe rclcpp unitree_go)
install(TARGETS go2w_native_state_probe DESTINATION lib/${PROJECT_NAME})
ament_package()
